\title{Pair Selection Reaches the Cram\'er--Rao Bound in Multi-Channel
Transient Localisation over WDM Fibres}

\begin{abstract}
A perturbation modulates the polarisation along a WDM fibre; chromatic
dispersion delays its signature across channels. We derive the
Cram\'er--Rao bound and show that selecting well-separated channel pairs
--- not fusing all pairs --- attains it.
\end{abstract}

\section{Introduction}
Optical fibre sensing localises mechanical transients (vibration, strain,
acoustic waves) along a link and supports infrastructure monitoring and
intrusion detection~\cite{taylor1993das,he2022das}. When the link carries a WDM
comb, a \emph{single} transient imprints a \emph{wavelength-dependent} delay:
chromatic dispersion (CD) makes the signature arrive at different times in
different channels. The inter-channel delay is proportional to the travelled
distance, so multi-channel time-delay estimation directly yields the range
$Z$~\cite{sadler2003survey}. With $\sim\!10$ channels, hundreds of channel pairs
are available, which naively suggests that fusing them must improve accuracy.

We show, at proof-of-concept level, that the essential step is \emph{pair
selection}. Naive fusion over \emph{all} channel pairs is dominated by
uninformative small-separation pairs; restricting to \emph{well-separated}
pairs attains the bound, and dispersion weighting is simply the soft-filter
equivalent. Our contributions are: (i)~the effective-SNR Cram\'er--Rao bound
(CRB) for CD-based transient localisation, validated against Monte Carlo;
(ii)~the result that equal-weight fusion over all pairs falls far from the bound
(and can be beaten by a single well-chosen pair), whereas selecting
well-separated pairs --- combined equally or by dispersion weighting --- attains
it; and (iii)~signal-to-noise-ratio (SNR)-free estimators plus the scaling laws
that turn this into a design rule.

\section{System Model and Cram\'er--Rao Bound}
A fixed input state of polarisation (SOP) propagates through a fibre with
polarisation-mode dispersion (PMD; random wave-plate model) and CD. A transient
of bandwidth $B$ modulates the local birefringence over a short section at
distance $Z$. We represent the received SOP by its 2-D Stokes projection
$(S_1,S_2)$ after rotating the centroid to the north pole. For channels at
wavelengths $\lambda_i,\lambda_j$ the CD delay is
\begin{equation}
\tau_{ij}=Z\,D_{ij},\quad
D_{ij}=\int_{\lambda_j}^{\lambda_i}\!D(\lambda)\,d\lambda,\quad
D(\lambda)=D_0+S_0(\lambda-\lambda_0),
\end{equation}
hence a pair-distance estimate is $\hat{Z}_{ij}=\hat{\tau}_{ij}/D_{ij}$.

For a discrete signal $x[n]$ with per-sample noise variance $\sigma^2$, the
\emph{energy} SNR is $\SNR=E/N_0=\sum_n x[n]^2/\sigma^2$, and the mean-square
bandwidth is $\beta^2=\sum_k(2\pi f_k)^2|X_k|^2/\sum_k|X_k|^2$. For two noisy
observations of a common signal, the delay information is governed by the
\emph{effective} SNR $\SNReff(f)=\rho_1\rho_2/(1+\rho_1+\rho_2)$, with
$\rho_i=S/N_i$; for equal sensors $\SNReff=\rho^2/(1+2\rho)$. The Fisher
information and CRB are
\begin{equation}
J_{ij}=\sum_k \omega_k^2\,\SNReff(f_k),\qquad
\CRB(\tau_{ij})=1/J_{ij},\qquad
\sigma_{Z,ij}=\frac{\sqrt{\CRB(\tau_{ij})}}{D_{ij}}.
\end{equation}
At high SNR $\SNReff\!\to\!\rho/2$ and $\CRB=1/(\SNR\beta^2)$: SNR and
\emph{bandwidth} are the fundamental limits. Equation~(2) is our reference
throughout.

\section{Multi-Channel Fusion Estimators}
From one estimate per channel pair, the distance is a weighted mean
$\hat Z=\sum_{ij}w_{ij}\hat Z_{ij}/\sum_{ij}w_{ij}$. We compare four
\emph{SNR-free} rules (only the known $D_{ij}$ is used):
\begin{itemize}\setlength\itemsep{1pt}
\item \textbf{Single pair}: one channel pair (best $=\max D$, worst $=\min D$).
\item \textbf{Average}: $w_{ij}=1$ (naive).
\item \textbf{Dispersion-weighted}: $w_{ij}=D_{ij}^2$, the minimum-variance weight
because $\sigma_{Z,ij}\!\propto\!1/D_{ij}$ when the per-pair CRB is common.
\item \textbf{Robust}: $w_{ij}=D_{ij}^2\,\psi(r_{ij})$ with a Tukey-bisquare
$\psi=(1-(r/(c\,s))^2)^2$ on residuals $r_{ij}=|\hat Z_{ij}-\med\hat Z|$,
$s=\mathrm{IQR}$; it suppresses threshold-region outliers.
\end{itemize}
Because pairs share channels, their errors are correlated; a strict bound
requires the full covariance (a structured GLS combination). We use Eq.~(2) as
a tight practical reference and note the correlation penalty as a caveat.

\section{Results}
We simulate $10$ independent PMD fibres, a C+L WDM comb (241 channels,
50~GHz), $B\!\in\![200\,$Hz$,2\,$MHz$]$, energy
SNR$\,\in\![15,100]\,$dB, $100$ realisations, and channel subsets of size
$4$--$12$ (extremes first); delays use parabolic interpolation of the 2-D
cross-correlation.

\begin{figure}[t]
\centering
\includegraphics[width=0.9\linewidth]{F1_rmse_vs_snr.png}
\caption{RMSE vs SNR at three bandwidths ($12$ channels), fusing \emph{all}
channel pairs (no selection). Equal-weight averaging (circles) is far above the
bound; dispersion-weighted (squares) and robust (diamonds) attain the CRB
(dashed, ratio $\approx\!1.0$--$1.1$).}
\label{fig:money}
\end{figure}

\textbf{Naive all-pairs fusion fails} (Fig.~\ref{fig:money}). At high SNR the dispersion-weighted and
robust rules lie on the CRB ($\mathrm{RMSE}/\mathrm{CRB}\approx1.0$--$1.1$),
whereas equal-weight averaging over \emph{all} pairs remains
$10$--$10^{3}\times$ above it. The reason follows from Eq.~(2):
$\sigma_{Z,ij}\!\propto\!1/D_{ij}$, so low-separation pairs carry almost no
information, yet equal weighting lets them dominate. The fix is pair selection.

\begin{figure}[t]
\centering
\includegraphics[width=\linewidth]{F7_pool_effect.png}
\caption{Pair selection is the essential step. Left: RMSE vs SNR ($12$ ch) for
equal-weight averaging over \emph{all} pairs (circles) and over a
\emph{filtered pool} with $D\!\ge\!0.25\max D$ (squares); on the filtered pool,
equal averaging, $D^2$-weighting and robust all coincide on the CRB. Right:
residual SNR saving of weighting over averaging --- $\sim\!30$\,dB on all pairs
but $\approx 0$ after filtering. Dispersion weighting is the soft-filter
equivalent of an explicit threshold.}
\label{fig:pool}
\end{figure}

\textbf{Pair selection, not weighting, is what matters} (Fig.~\ref{fig:pool}).
Restricting the pool to well-separated pairs ($D\ge 0.25\max D$) removes the
uninformative pairs and the naive average drops onto the CRB; the residual
benefit of dispersion weighting is then $\approx 0$\,dB. In other words,
weighting and hard filtering are two implementations of the same idea ---
discard low-separation pairs. Because the retained pairs have similar
precision, \emph{any} sensible combination reaches the bound, which makes the
proposed scheme calibration-free and robust to the exact weighting.

\textbf{Design rules.} (i)~At high SNR the RMSE scales as $1/(\sqrt{\SNR}\,B)$
(Eq.~(2)), so the perturbation \emph{bandwidth is as fundamental as SNR}.
(ii)~The SNR penalty of fusing \emph{all} pairs, relative to selecting a
well-separated pool, is $\sim$28--35\,dB, and is largest for few channels.
(iii)~The SNR required to reach RMSE$\,=5\,$km falls monotonically with the
number of channels ($\sim$62--67\,dB for the selected pool versus
$\sim$92--100\,dB for all-pairs averaging). (iv)~Averaging $N$ independent fibres
lowers the RMSE as $1/\sqrt{N}$, independently of the fusion rule. (v)~The robust
rule departs from the CRB only in the threshold region, reducing heavy-tail
error. These curves are provided as supplementary material.

\section{Conclusion}
We derived and validated the effective-SNR CRB for CD-based transient
localisation and showed that the essential step is \emph{pair selection}: fusing
\emph{all} channel pairs is dominated by uninformative small-separation pairs,
whereas restricting to well-separated pairs attains the bound regardless of the
combining rule --- dispersion weighting being the soft-filter equivalent. The
scheme needs no SNR knowledge, so it is directly deployable, and because the
bound scales as $1/(\sqrt{\SNR}\,B)$, slow perturbations are the hard case.
Future work: experimental validation, covariance-aware fusion, and optimisation
of the channel-selection scheme.

\section{Equation reference (working notes --- not for submission)}
\noindent This page collects every relation used in the study, with a short
explanation of each. Units are indicated.

\smallskip
\noindent\textbf{1) Chromatic-dispersion delay and distance.}
\begin{equation}
\tau_{ij}=Z\,D_{ij},\qquad
D_{ij}=\int_{\lambda_j}^{\lambda_i}\!(D_0+S_0(\lambda-\lambda_0))\,d\lambda,
\qquad \hat Z_{ij}=\hat\tau_{ij}/D_{ij}.
\end{equation}
$D(\lambda)$ is the dispersion curve [ps/(nm\,km)], $D_{ij}=|\!\int D\,d\lambda|$
[ps/km], $Z$ [km], $\tau_{ij}$ [s]; the estimate inverts the known dispersion
integral.

\smallskip
\noindent\textbf{2) Energy SNR and target noise.}
\begin{equation}
\SNR=\frac{E}{N_0}=\frac{\sum_n x[n]^2}{\sigma^2},\qquad
\sigma^2=\frac{\sum_n x[n]^2}{\SNR_{\rm lin}} \quad (N_0=\sigma^2\,dt).
\end{equation}
$E=\int x^2 dt$ is signal energy, $N_0$ the two-sided noise PSD; the last form
sets the per-component noise variance for a target SNR.

\smallskip
\noindent\textbf{3) Mean-square bandwidth.}
\begin{equation}
\beta^2=\frac{\int (2\pi f)^2|S(f)|^2\,df}{\int|S(f)|^2\,df}
=\frac{\sum_k (2\pi f_k)^2|X_k|^2}{\sum_k |X_k|^2}.
\end{equation}
$S(f)$ is the signal spectrum; $\beta$ is the RMS bandwidth that sets the
delay precision.

\smallskip
\noindent\textbf{4) Deterministic (known-signal) CRB.}
\begin{equation}
\CRB_{\det}(\tau)=\frac{1}{\SNR\,\beta^2}.
\end{equation}
The single-sensor ideal bound; the two-sensor effective bound sits a factor
$\sim\!2$ above it at high SNR (reference-noise penalty).

\smallskip
\noindent\textbf{5) Two-sensor effective SNR.}
\begin{equation}
\SNReff(f)=\frac{(S/N_1)(S/N_2)}{1+S/N_1+S/N_2},\qquad
\SNReff=\frac{\rho^2}{1+2\rho}\ \ (\text{equal},\ \rho=S/N_0).
\end{equation}
Combines the two sensors' SNRs \emph{before} cross-correlation; it is not the
correlator-output SNR.

\smallskip
\noindent\textbf{6) Fisher information and delay CRB.}
\begin{equation}
J_{ij}=\sum_k \omega_k^2\,\SNReff(f_k),\qquad
\CRB(\tau_{ij})=1/J_{ij},\qquad
\sigma_{Z,ij}=\sqrt{\CRB(\tau_{ij})}\,\frac{1}{D_{ij}}.
\end{equation}
Discrete one-sided sum ($\omega_k=2\pi f_k$); the last relation maps delay
variance [s$^2$] to distance std [km] (with $D_{ij}$ in ps/km and $1\leftrightarrow10^{12}$ ps/s).

\smallskip
\noindent\textbf{7) Per-channel variance model.}
\begin{equation}
\Var(\tau_{ij})=u_i+u_j,\qquad u_k=1/J_k,
\end{equation}
with $J_k$ the single-channel Fisher information. Additivity of the pairwise
variance in the per-channel terms motivates the $D^2$ weighting.

\smallskip
\noindent\textbf{8) Fusion estimators (SNR-free).}
\begin{equation}
\hat Z^{\rm avg}\!=\!\frac1{n_p}\!\sum_{ij}\!\hat Z_{ij},\quad
\hat Z^{D^2}\!=\!\frac{\sum_{ij}D_{ij}^2\hat Z_{ij}}{\sum_{ij}D_{ij}^2},\quad
w_{ij}=D_{ij}^2\,\psi(r_{ij}).
\end{equation}
$\psi$ is the Tukey-bisquare; the second rule is inverse-variance weighting when
$\sigma_{Z,ij}\!\propto\!1/D_{ij}$.

\smallskip
\noindent\textbf{9) Tukey-bisquare robust weight.}
\begin{equation}
\psi(r)=\Bigl[1-\bigl(r/(c\,s)\bigr)^2\Bigr]^2\ \ (|r|<c\,s),\quad 0\ \text{otherwise},
\qquad r_{ij}=|\hat Z_{ij}-\med\hat Z|,\ s=\mathrm{IQR}.
\end{equation}
Smooth down-weighting of outliers; $c\!\approx\!2.5$ with $s\!=\!1.5\,\mathrm{IQR}$
(cutoff $\approx3.75\,$IQR) is used here.

\smallskip
\noindent\textbf{10) Correlation-aware (GLS) combination.}
\begin{equation}
\Sigma_\tau=A\,\mathrm{diag}(u)\,A^{T},\qquad
\Sigma_Z=\mathrm{diag}(1/D)\,\Sigma_\tau\,\mathrm{diag}(1/D),\qquad
w=\Sigma_Z^{+}\mathbf 1,\qquad
\Var(\hat Z)=\frac{1}{\mathbf 1^{T}\Sigma_Z^{+}\mathbf 1}.
\end{equation}
$A$ is the signed incidence matrix of the pairs ($A_{k,i}=+1,\,A_{k,j}=-1$);
$\Sigma_\tau$ has rank $n_{\rm ch}-1$, encoding the fact that only $n_{\rm ch}-1$
delays are independent. This is the covariance-optimal fusion.

\smallskip
\noindent\textbf{11) Combined CRB and scaling.}
\begin{equation}
\sigma_{Z,\rm comb}=\Bigl[\sum_{ij} D_{ij}^2/\bigl(\CRB(\tau_{ij})\cdot10^{24}\bigr)\Bigr]^{-1/2},
\qquad
\sigma_Z \propto \frac{1}{\sqrt{\SNR}\,\beta}\xrightarrow{\beta\propto B}\frac{1}{\sqrt{\SNR}\,B},
\qquad
\sigma_Z\propto \frac{1}{\sqrt{N_{\rm fib}}}.
\end{equation}
The first is inverse-variance combination of pair CRBs; the last two are the
high-SNR and multi-fibre scaling laws.

\smallskip
\noindent\textbf{12) Performance metrics.}
\begin{equation}
\mathrm{RMSE}=\sqrt{\tfrac1M\textstyle\sum_m(\hat Z_m-Z)^2},\quad
\mathrm{bias}=\tfrac1M\textstyle\sum_m(\hat Z_m-Z),\quad
\mathrm{success}=\tfrac1M\#\{|\hat Z_m-Z|\le \epsilon\}.
\end{equation}
$\mathrm{MSE}=\mathrm{bias}^2+\mathrm{std}^2$; $\epsilon$ is the localisation
tolerance (here $1\%$ of $Z$). Reported over all fibres and realisations.
