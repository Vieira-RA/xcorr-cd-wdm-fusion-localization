\title{How to Reach the Cram\'er--Rao Bound in Multi-Channel Transient
Localisation over WDM Fibres: Pair Selection, Robust Fusion, and Estimator Bias}

\author{Author One,\authormark{1} Author Two,\authormark{1,*} and Author Three\authormark{2}}

\address{\authormark{1}Group, Department, University, City, Postcode, Country\\

\begin{abstract}
A transient perturbation modulates the state of polarisation along a WDM
fibre; chromatic dispersion makes its signature arrive at different times
across channels, so the inter-channel delay yields the perturbation distance.
We derive the two-sensor effective-SNR Cram\'er--Rao bound (CRB) for this
problem and study, end to end, the three practical levers that decide how close
a real estimator gets to it: (i)~which channel pairs to use, (ii)~how to fuse
them, and (iii)~which delay estimator to apply. We show that \emph{pair
selection} is the essential step (dispersion weighting is its soft-filter
equivalent); that a \emph{robust} (bisquare) combiner is best on a realistic
equispaced channel grid; and that the widely used parabolic peak estimator is
variance-optimal but \emph{biased}, so an unbiased alternative (GCC-PHAT) wins
at high SNR --- a regime-dependent trade-off we quantify. We report the scaling
laws in SNR, bandwidth, channel count and fibre count. The result is a
rigorous bound together with a practical recipe.
\end{abstract}

\section{Introduction}
\label{sec:intro}
Distributed optical sensing localises mechanical perturbations (vibration,
strain, acoustic transients) along a fibre and underpins applications from
infrastructure monitoring to intrusion detection~\cite{taylor1993das,he2022das}.
When the link carries a WDM comb, a \emph{single} transient imprints a
\emph{wavelength-dependent} delay: chromatic dispersion (CD) makes the signature
arrive at different times in different channels. The inter-channel delay is
proportional to the travelled distance, so multi-channel time-delay estimation
directly yields the range $Z$~\cite{sadler2003survey}. With $\sim\!10$ channels,
hundreds of channel pairs are available, which naively suggests that fusing them
must improve accuracy.

This paper asks a practical question: \emph{given a bound, how do we reach it?}
We identify three levers --- pair selection, fusion rule, and delay estimator ---
and quantify each with an end-to-end PMD/CD simulation.

\smallskip
\noindent\textbf{Contributions.}
\begin{enumerate}\setlength\itemsep{1pt}
\item \textbf{The bound.} We derive the two-sensor \emph{effective-SNR}
Cram\'er--Rao bound for CD-based transient localisation
(Sec.~\ref{sec:crb}) and validate it: the per-pair bound matches Monte-Carlo to
within $\sim\!10\%$.
\item \textbf{Pair selection $\equiv$ weighting} (Sec.~\ref{sec:fusion},
Fig.~\ref{fig:pool}). Fusing \emph{all} channel pairs with equal weight is
catastrophic (dominated by uninformative small-separation pairs). Restricting to
\emph{well-separated} pairs attains the bound, after which equal, dispersion- and
robust-weighted fusion coincide --- dispersion weighting is simply the
soft-filter equivalent of an explicit threshold.
\item \textbf{Robust fusion wins on a realistic grid} (Sec.~\ref{sec:fusion},
Fig.~\ref{fig:alpha}). On an equispaced channel grid a Tukey-bisquare combiner
gives the lowest RMSE across the plane.
\item \textbf{Estimator bias limits high SNR} (Sec.~\ref{sec:estimators},
Fig.~\ref{fig:estimators}). The parabolic peak interpolator is variance-optimal
but carries a systematic bias that dominates at high SNR and low bandwidth;
GCC-PHAT is unbiased and wins there --- a \emph{regime-dependent} trade-off.
\item \textbf{Scaling laws} (Sec.~\ref{sec:results}): $\sigma_Z\propto
1/(\sqrt{\SNR}\,B)$, monotone improvement with channel count, and
$1/\sqrt{N_{\rm fib}}$ across independent fibres.
\end{enumerate}

\section{System model}
\label{sec:model}
\textbf{Polarisation and the Jones/Stokes picture.} A fixed input state of
polarisation (SOP) $\hat s_{\rm in}$ propagates through the fibre. The link is
described by a frequency-dependent unitary Jones matrix $U(\omega)$; the output
Stokes vector is the rotation
\begin{equation}
\mathbf s_{\rm out}(\omega)=R\big(U(\omega)\big)\,\mathbf s_{\rm in},
\end{equation}
where $R(U)$ is the $3\times3$ rotation induced by $U$ (via the Pauli matrices).
We represent the received SOP by its 2-D Stokes projection $(S_1,S_2)$ after
rotating the centroid to the north pole, which removes the static SOP offset.

\smallskip
\noindent\textbf{Fibre propagation (PMD + CD).} Birefringence is modelled with a
random wave-plate model: the fibre is a cascade of segments of length $L_F$ with
random orientation and a birefringence vector
$\boldsymbol\beta(z)=\boldsymbol\beta_0(z)+\Delta\omega\,\boldsymbol\beta'(z)$,
so that $dU/dz=-\tfrac{i}{2}[\boldsymbol\beta(z)\!\cdot\!\boldsymbol\sigma]U$
is integrated along $z$. This produces a realistic frequency autocorrelation
(i.e. PMD) between channels. Chromatic dispersion is applied separately as a
deterministic, channel-dependent group delay.

\smallskip
\noindent\textbf{Transient and its bandwidth.} A transient of (3-dB) bandwidth
$B$ modulates the local birefringence over a short section at distance $Z$
through a Gaussian envelope
\begin{equation}
s_{\rm env}(t)=1+A_{\rm p}\exp\!\Big[-\frac{(t-t_0)^2}{2\sigma_t^2}\Big],
\qquad \sigma_t=\frac{0.3748}{B}.
\label{eq:env}
\end{equation}
The perturbation thus appears as a transient bump in the SOP; its
\emph{bandwidth} $B$ sets how fast the signature evolves and therefore the
achievable timing precision.

\smallskip
\noindent\textbf{CD delay $\to$ distance.} For channels at wavelengths
$\lambda_i,\lambda_j$ the CD delay is
\begin{equation}
\tau_{ij}=Z\,D_{ij},\qquad
D_{ij}=\int_{\lambda_j}^{\lambda_i}\! D(\lambda)\,d\lambda,\qquad
D(\lambda)=D_0+S_0(\lambda-\lambda_0),
\label{eq:cd}
\end{equation}
hence a pair-distance estimate is $\hat Z_{ij}=\hat\tau_{ij}/D_{ij}$. The
dispersion integral $D_{ij}$ is a \emph{known} fibre property, so pairs with a
large $|D_{ij}|$ are a-priori identifiable.

\section{Cram\'er--Rao bound}
\label{sec:crb}
\textbf{Energy SNR.} For a discrete signal $x[n]$ with per-sample noise
variance $\sigma^2$, the \emph{energy} SNR is
\begin{equation}
\SNR=\frac{E}{N_0}=\frac{\sum_n x[n]^2}{\sigma^2},
\qquad N_0=\sigma^2\,dt,
\end{equation}
where $E=\int x^2\,dt$ and $N_0$ is the two-sided noise PSD. To realise a target
SNR we inject Gaussian noise with variance $\sigma^2=\sum_n x[n]^2/\SNR_{\rm lin}$.

\smallskip
\noindent\textbf{Mean-square bandwidth.} The RMS bandwidth that sets the delay
precision is
\begin{equation}
\beta^2=\frac{\int (2\pi f)^2|S(f)|^2\,df}{\int|S(f)|^2\,df}
=\frac{\sum_k (2\pi f_k)^2|X_k|^2}{\sum_k |X_k|^2}.
\label{eq:beta}
\end{equation}

\smallskip
\noindent\textbf{Two-sensor effective SNR.} For two noisy observations of a
common signal the delay information is governed by the \emph{effective} SNR,
\begin{equation}
\SNReff(f)=\frac{(S/N_1)(S/N_2)}{1+S/N_1+S/N_2},
\qquad
\SNReff=\frac{\rho^2}{1+2\rho}\ \ (\text{equal sensors},\ \rho=S/N_0).
\label{eq:effsnr}
\end{equation}
This combines the two sensors' SNRs \emph{before} any correlation; it is not the
cross-correlation SNR.

\smallskip
\noindent\textbf{Fisher information and the CRB.}
\begin{equation}
J_{ij}=\sum_k \omega_k^2\,\SNReff(f_k),\qquad
\CRB(\tau_{ij})=1/J_{ij},\qquad
\sigma_{Z,ij}=\sqrt{\CRB(\tau_{ij})}\,\frac{1}{D_{ij}} .
\label{eq:crb}
\end{equation}
At high SNR $\SNReff\!\to\!\rho/2$, so $\CRB\!\to\!1/(\SNR\beta^2)$: \emph{SNR
and bandwidth are the fundamental limits.}

\smallskip
\noindent\textbf{Combining pairs: naive vs.\ star vs.\ GLS.} The naive
inverse-variance combination assumes the pair estimates are independent;
because all pairs share channels they are \emph{correlated} (only $n_{\rm ch}-1$
delays are independent), so it is optimistic. The achievable reference is the
\emph{star} combination: pick one reference channel and keep only the
$n_{\rm ch}-1$ pairs connecting it to the others,
\begin{equation}
\sigma_{Z,\rm star}=\Bigl[\max_r \sum_{k\neq r} D_{rk}^2\big/(\CRB(\tau_{rk})\cdot10^{24})\Bigr]^{-1/2}.
\label{eq:star}
\end{equation}
A covariance-aware GLS variant (structured $\Sigma_\tau=A\,\mathrm{diag}(u)A^{T}$)
is tighter still but its per-channel variance fit degrades on wide-$D$ grids; we
use the star bound as the robust overlay.

\section{Multi-channel fusion}
\label{sec:fusion}
From one estimate per channel pair, the distance is a weighted mean
$\hat Z=\sum_{ij}w_{ij}\hat Z_{ij}/\sum_{ij}w_{ij}$. We consider:
\begin{itemize}\setlength\itemsep{1pt}
\item \textbf{Equal}: $w_{ij}=1$ (naive).
\item \textbf{Dispersion-weighted}: $w_{ij}=D_{ij}^2$ --- the minimum-variance
weight, because $\sigma_{Z,ij}\propto 1/D_{ij}$ when the per-pair CRB is common.
\item \textbf{Robust}: $w_{ij}=D_{ij}^2\,\psi(r_{ij})$ with a Tukey-bisquare
$\psi(r)=[1-(r/(c s))^2]^2$ on residuals $r_{ij}=|\hat Z_{ij}-\med\hat Z|$,
$s=\mathrm{IQR}$, which down-weights threshold-region outliers.
\end{itemize}
All are \textbf{SNR-free}: only the known $D_{ij}$ is required.

\smallskip
\noindent\textbf{Pair selection.} We restrict the pool to well-separated pairs,
$D_{ij}\ge \alpha\max_{ij}D_{ij}$, and study the residual value of weighting as
$\alpha$ grows (Fig.~\ref{fig:alpha}).

\section{Delay estimators}
\label{sec:estimators}
For each pair we estimate the delay from the 2-D cross-correlation. Two
estimators matter:

\smallskip
\noindent\textbf{Parabolic peak.} The magnitude cross-correlation peak is refined
by a three-point parabola. It is \emph{variance-optimal} in the CRB regime but,
for an asymmetric peak (channels with different PMD), it carries a systematic
\emph{bias} that grows as $B$ falls.

\smallskip
\noindent\textbf{GCC-PHAT.} Phase-transform generalised cross-correlation with a
bandwidth mask; its peak is sharper and its delay estimate is essentially
\emph{unbiased} for a pure delay, at the cost of higher variance.

\smallskip
\noindent The measured trade-off (15 ch, robust pool, RMSE in km) is summarised
in Table~\ref{tab:est} and Fig.~\ref{fig:estimators}. At 22\,kHz/100\,dB the
unbiased GCC-PHAT is $3.7\times$ more accurate; at 1.3\,kHz/76\,dB the
lower-variance parabolic wins. The choice is therefore \emph{regime-dependent}.

\section{Simulation setup}
\label{sec:setup}
We simulate $10$ independent PMD fibres. Two channel grids are studied: a
\emph{band-edge} selection (extremes first) and a realistic \emph{equispaced}
grid (custom 30-channel master over 184--196\,THz, uniform subsets of
$\{5,10,15\}$ channels). Sweeps: bandwidth
$B\!\in\![0.2,2000]\,$kHz (log) and energy SNR $\in[15,100]\,$dB, $100$ noise
realisations per cell. Delay estimation uses parabolic interpolation or
GCC-PHAT; distances are combined with the rules of Sec.~\ref{sec:fusion}.

\section{Results}
\label{sec:results}
\textbf{The bound is attainable at the pair level.} The empirical per-pair
delay std matches the theoretical CRB to within $\sim\!10\%$ (ratio
$1.08$--$1.11$), confirming Eq.~\eqref{eq:crb}. On the combined estimate, the
star bound is coherent (simulation above it everywhere), whereas the naive
combination is optimistic.

\textbf{Selection dominates weighting} (Fig.~\ref{fig:pool}). Equal-weight
averaging over all pairs is $10$--$10^3\times$ above the CRB; on a filtered pool
the combiners coincide on the bound. On the equispaced grid the weighting
benefit is a few dB at $\alpha=0$ and shrinks with filtering
(Fig.~\ref{fig:alpha}).

\textbf{Robust fusion is best on the equispaced grid.} High-SNR RMSE (averaged
over bandwidth) for 15 ch: equal $6.60$, $D^2$ $6.25$, robust $\mathbf{5.48}$\,km
(CRB $2.39$); robust wins in $>$90\% of cells. On the band-edge grid the pool is
nearly homogeneous and the three rules coincide.

\textbf{Estimator bias limits high SNR} (Fig.~\ref{fig:estimators},
Table~\ref{tab:est}). The parabolic bias ($-2.6$\,km at 22\,kHz) dominates at
high SNR; GCC-PHAT removes it, at the price of an earlier threshold.

\textbf{Scaling laws.} (i) At high SNR $\mathrm{RMSE}\propto1/(\sqrt{\SNR}\,B)$
(Fig.~\ref{fig:estimators}). (ii) More channels lower the RMSE and the SNR
required to reach a target (Fig.~\ref{fig:design}). (iii) Averaging $N$
independent fibres reduces the RMSE as $1/\sqrt{N}$.

\section{Discussion and design rules}
\label{sec:discussion}
From the results, a practical recipe:
\begin{enumerate}\setlength\itemsep{1pt}
\item \textbf{Select well-separated channel pairs} (large $|D_{ij}|$); this is
the dominant lever, and dispersion weighting is its soft-filter equivalent.
\item \textbf{Fuse with a robust (bisquare) combiner} --- it is never worse than
equal or $D^2$ weighting and is best on a realistic equispaced grid.
\item \textbf{Choose the delay estimator by regime}: parabolic at low SNR or low
bandwidth (variance-limited); GCC-PHAT at high SNR / high bandwidth
(bias-limited). A bias-corrected parabolic would combine both advantages.
\item Budget for the fundamental limits: precision improves as
$1/(\sqrt{\SNR}\,B)$ and as more channels / fibres are averaged.
\end{enumerate}
\textbf{Limitations.} The study is simulation-only; it assumes a single
perturbation and a static dispersion curve. The parabolic bias is a property of
the interpolator and is not corrected here; we present it as a trade-off.

\section{Conclusion}
\label{sec:conclusion}
We derived and validated the effective-SNR CRB for CD-based transient
localisation and studied the three levers that decide whether a real estimator
reaches it: pair selection (essential), robust fusion (best on a realistic
grid), and the delay estimator (a bias/variance trade-off --- parabolic at low
SNR/low bandwidth, GCC-PHAT at high SNR/high bandwidth). The recipe needs no SNR
knowledge, so it is directly deployable. Future work: experimental validation,
bias-corrected peak estimation, and optimisation of the channel-selection
scheme.

\smallskip
\noindent\textbf{1) CD delay and distance.}
\begin{equation}
\tau_{ij}=Z\,D_{ij},\quad
D_{ij}=\int_{\lambda_j}^{\lambda_i}(D_0+S_0(\lambda-\lambda_0))\,d\lambda,\quad
\hat Z_{ij}=\hat\tau_{ij}/D_{ij}.
\end{equation}

\smallskip
\noindent\textbf{2) Energy SNR and target noise.}
\begin{equation}
\SNR=\frac{E}{N_0}=\frac{\sum_n x[n]^2}{\sigma^2},\qquad
\sigma^2=\frac{\sum_n x[n]^2}{\SNR_{\rm lin}}\quad(N_0=\sigma^2 dt).
\end{equation}

\smallskip
\noindent\textbf{3) Mean-square bandwidth.}
\begin{equation}
\beta^2=\frac{\sum_k(2\pi f_k)^2|X_k|^2}{\sum_k|X_k|^2}.
\end{equation}

\smallskip
\noindent\textbf{4) Deterministic CRB.}
\begin{equation}
\CRB_{\det}(\tau)=1/(\SNR\,\beta^2).
\end{equation}

\smallskip
\noindent\textbf{5) Two-sensor effective SNR.}
\begin{equation}
\SNReff(f)=\frac{(S/N_1)(S/N_2)}{1+S/N_1+S/N_2},\qquad
\SNReff=\rho^2/(1+2\rho).
\end{equation}

\smallskip
\noindent\textbf{6) Fisher information and delay CRB.}
\begin{equation}
J_{ij}=\sum_k \omega_k^2\,\SNReff(f_k),\qquad
\CRB(\tau_{ij})=1/J_{ij},\qquad
\sigma_{Z,ij}=\sqrt{\CRB(\tau_{ij})}/D_{ij}.
\end{equation}

\smallskip
\noindent\textbf{7) Per-channel variance model.}
\begin{equation}
\Var(\tau_{ij})=u_i+u_j,\qquad u_k=1/J_k.
\end{equation}

\smallskip
\noindent\textbf{8) Fusion estimators (SNR-free).}
\begin{equation}
\hat Z^{\rm avg}=\frac1{n_p}\!\sum_{ij}\hat Z_{ij},\quad
\hat Z^{D^2}=\frac{\sum_{ij}D_{ij}^2\hat Z_{ij}}{\sum_{ij}D_{ij}^2},\quad
w_{ij}=D_{ij}^2\,\psi(r_{ij}).
\end{equation}

\smallskip
\noindent\textbf{9) Tukey-bisquare robust weight.}
\begin{equation}
\psi(r)=[1-(r/(c s))^2]^2\ \ (|r|<cs),\quad 0\ \text{otherwise},\quad
r_{ij}=|\hat Z_{ij}-\med\hat Z|,\ s=\mathrm{IQR}.
\end{equation}

\smallskip
\noindent\textbf{10) Star and GLS combination.}
\begin{equation}
\sigma_{Z,\rm star}=\Bigl[\max_r\sum_{k\neq r}D_{rk}^2/(\CRB(\tau_{rk})10^{24})\Bigr]^{-1/2},
\qquad
\Sigma_\tau=A\,\mathrm{diag}(u)A^{T},\ w=\Sigma_Z^{+}\mathbf 1.
\end{equation}

\smallskip
\noindent\textbf{11) Scaling and metrics.}
\begin{equation}
\sigma_Z\propto\frac{1}{\sqrt{\SNR}\,\beta},\quad
\sigma_Z\propto\frac{1}{\sqrt{N_{\rm fib}}},\quad
\mathrm{RMSE}=\sqrt{\tfrac1M\textstyle\sum_m(\hat Z_m-Z)^2},\quad
\MSE=\mathrm{bias}^2+\mathrm{std}^2.
\end{equation}
